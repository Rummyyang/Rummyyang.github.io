( set -u
# H800: github.com timed out during the setup. Find a working proxy setting on this machine, save it to /root/h800_proxy.env,
# and restart the setup + loop watcher with it. Read-only except that one env file and the tmux restarts. No proxy value is printed.
OLD=/m2v_intern/liuwenze/code/opd-run
C=/root/h800_proxy_candidates.txt; : > $C
{ for f in /etc/environment /etc/profile /etc/profile.d/*.sh /root/.bashrc /root/.profile /root/.bash_profile /home/*/.bashrc /home/*/.profile $OLD/*.env $OLD/env.sh $OLD/scripts/env.sh; do [ -f "$f" ] && grep -hoE "(https?_proxy|HTTPS_PROXY|HTTP_PROXY|all_proxy|ALL_PROXY)=['\"]?[^'\" ]+" "$f" 2>/dev/null; done
  for p in 1 $(pgrep -f "sshd|tmux|bash" | head -20); do tr '\0' '\n' < /proc/$p/environ 2>/dev/null | grep -E "^(https?_proxy|HTTPS_PROXY|HTTP_PROXY|all_proxy|ALL_PROXY)="; done
  env | grep -E "^(https?_proxy|HTTPS_PROXY|HTTP_PROXY|all_proxy|ALL_PROXY)="; } | sed "s/['\"]//g" | sed -E 's/^[a-zA-Z_]+=//' | sort -u > $C
echo "proxy candidates found: $(wc -l < $C) (values not shown)"
GOOD=""
while read -r u; do [ -n "$u" ] || continue; code=$(https_proxy=$u HTTPS_PROXY=$u curl -sS -o /dev/null -w "%{http_code}" --max-time 20 https://github.com/ 2>/dev/null); hf=$(https_proxy=$u HTTPS_PROXY=$u curl -sS -o /dev/null -w "%{http_code}" --max-time 20 https://huggingface.co/api/models/Qwen/Qwen3-1.7B 2>/dev/null); echo "  candidate $(echo "$u" | cut -c1-8)... github $code hf $hf"; [ "$code" = 200 ] && [ "$hf" = 200 ] && { GOOD=$u; break; }; done < $C
echo "direct: github $(curl -sS -o /dev/null -w '%{http_code}' --max-time 15 https://github.com/ 2>/dev/null) hf $(curl -sS -o /dev/null -w '%{http_code}' --max-time 15 https://huggingface.co/api/models/Qwen/Qwen3-1.7B 2>/dev/null)"
if [ -z "$GOOD" ]; then echo "NO WORKING PROXY FOUND: send this output; also ask the machine owner how git/pip reach the internet here (proxy host:port)"; exit 1; fi
NP="localhost,127.0.0.1,::1,$(hostname),$(hostname -I 2>/dev/null | tr -s ' ' ',' | sed 's/,$//')"
umask 077; printf 'export https_proxy=%s\nexport HTTPS_PROXY=%s\nexport http_proxy=%s\nexport HTTP_PROXY=%s\nexport no_proxy=%s\nexport NO_PROXY=%s\n' "$GOOD" "$GOOD" "$GOOD" "$GOOD" "$NP" "$NP" > /root/h800_proxy.env
echo "saved /root/h800_proxy.env (github and hf reachable through it)"
tmux kill-session -t h800 2>/dev/null; tmux kill-session -t h800loops 2>/dev/null
tmux new-session -d -s h800 '. /root/h800_proxy.env; bash /root/h800_setup.sh 2>&1 | tee -a /root/h800_setup.log; echo EXIT_$? >> /root/h800_setup.log; sleep infinity'
tmux new-session -d -s h800loops '. /root/h800_proxy.env; TOKEN_FILE=/root/.hf_token_opd bash /root/h800_loops.sh 2>&1 | tee -a /root/h800_loops.log; sleep infinity'
sleep 40; echo "== setup (last lines) =="; tail -n 8 /root/h800_setup.log; echo "== loops =="; tail -n 2 /root/h800_loops.log )
