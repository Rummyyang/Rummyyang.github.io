def procs():
    out = {}
    for p in os.listdir("/proc"):
        if p.isdigit():
            try:
                st = open(f"/proc/{p}/stat").read().rsplit(")", 1)[1].split()
                out[int(p)] = (int(st[1]), int(st[3]), open(f"/proc/{p}/cmdline", "rb").read().replace(b"\0", b" ").decode(errors="replace"))
