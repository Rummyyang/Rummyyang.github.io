( set -u
T=/code/opd; O=$T/run; RB=$T/runbook; TOKEN_FILE=/root/.hf_token_opd
echo "== C1. stop the self-test and the old setup windows (nothing else of ours may start loops later) =="
tmux kill-session -t h800fix 2>/dev/null; tmux kill-session -t h800 2>/dev/null; tmux kill-session -t h800loops 2>/dev/null
pkill -TERM -f "^bash /root/h800_(setup|loops|fix)\.sh|scripts/(verify|run_arm|overnight|cold_start)\.sh" 2>/dev/null
for p in $(nvidia-smi --query-compute-apps=pid --format=csv,noheader); do kill -TERM $p 2>/dev/null; done
pkill -TERM -f "opd-env/|G-OPD/|/code/opd/|ray_gpu" 2>/dev/null; sleep 5; pkill -KILL -f "opd-env/|G-OPD/|/code/opd/|ray_gpu" 2>/dev/null
for p in $(pgrep -f "^ray::"); do tr '\0' '\n' < /proc/$p/environ 2>/dev/null | grep -q "ray_gpu" && kill -KILL $p 2>/dev/null; done
rm -rf /tmp/ray_gpu* 2>/dev/null
for i in $(seq 1 6); do G=$(nvidia-smi --query-compute-apps=pid --format=csv,noheader | grep -c .); [ "$G" = 0 ] && break; sleep 10; done
echo "GPU processes: $G; our loops running: $(pgrep -fc 'scripts/(heartbeat|run_tasks).sh')"
[ "$G" = 0 ] || { echo "STOP: GPU processes still there"; nvidia-smi --query-compute-apps=pid,used_memory --format=csv; exit 1; }
echo "== C2. prerequisites =="
[ -s "$TOKEN_FILE" ] || { echo "STOP: token file missing"; exit 1; }
[ -f /root/h800_proxy.env ] || { echo "STOP: /root/h800_proxy.env missing"; exit 1; }
. /root/h800_proxy.env
[ -d $RB/scripts ] && [ -f $O/paths.env ] || { echo "STOP: runbook or paths.env missing"; exit 1; }
git -C $RB pull -q --ff-only || echo "note: git pull failed now; the loop retries every 60 s"
echo "runbook $(git -C $RB log -1 --format=%h); node $(sed -n 's/^export OPD_NODE=//p' $O/paths.env)"
echo "== C3. settings: dispatch frozen, 80 GB tier 2, main venv until flash-attn is built remotely, heartbeat every 120 s =="
cd $RB
for t in $(awk '$1 !~ /^#/ && $1 != "version:" && NF > 2 {for (i = 3; i <= NF; i++) {split($i, a, ":"); print a[1]}}' tasks/queue.txt | sort -u); do mkdir -p $O/runs/$t; [ -f $O/runs/$t/.done ] || [ -f $O/runs/$t/.paused ] || echo "seq 0 $(date -u +%FT%TZ) (first start: dispatch frozen)" > $O/runs/$t/.paused; done
touch $O/control.env
sed -i '/^export VENV=/d' $O/control.env
grep -qx 'export PREFETCH_PAR=0' $O/control.env || echo 'export PREFETCH_PAR=0' >> $O/control.env
grep -q '^export FIXED_MICRO_BSZ=' $O/control.env && sed -i 's/^export FIXED_MICRO_BSZ=.*/export FIXED_MICRO_BSZ=1/' $O/control.env || echo 'export FIXED_MICRO_BSZ=1' >> $O/control.env
grep -q '^export VLLM_UTIL=' $O/control.env && sed -i 's/^export VLLM_UTIL=.*/export VLLM_UTIL=0.35/' $O/control.env || echo 'export VLLM_UTIL=0.35' >> $O/control.env
grep -q '^export GC=' $O/control.env || echo 'export GC=1' >> $O/control.env
mkdir -p $O/control $O/logs; echo 120 > $O/control/heartbeat_sec
echo "control.env: $(sed -n 's/^export //p' $O/control.env | tr '\n' ' '); paused $(ls $O/runs/*/.paused 2>/dev/null | wc -l)"
echo "== C4. start the loops (heartbeat + git pull/control + uploader), nothing dispatches =="
export OPD_ROOT=$O HF_REPO=yang31210999/opd-lab-b300-pub HF_TOKEN=$(head -1 $TOKEN_FILE | tr -d '[:space:]')
export XDG_CACHE_HOME=$T/cache UV_CACHE_DIR=$T/cache/uv UV_PYTHON_INSTALL_DIR=$T/tools/python PIP_CACHE_DIR=$T/cache/pip \
       HF_HOME=$T/cache/hf TRITON_CACHE_DIR=$T/cache/triton PATH=$T/tools/uv/bin:$PATH OVERNIGHT_TIER_A_ONLY=1 FA_CUDA_ARCHS=90 \
       PYTHONDONTWRITEBYTECODE=1 VLLM_NO_USAGE_STATS=1
setsid nohup bash scripts/heartbeat.sh > $O/logs/heartbeat.log 2>&1 < /dev/null &
setsid nohup bash scripts/run_tasks.sh > $O/logs/run_tasks.log 2>&1 < /dev/null &
sleep 90
echo "== C5. result =="
echo "loops $(pgrep -fc 'scripts/(heartbeat|run_tasks).sh'), started tasks $(grep -c '^START' $O/logs/run_tasks.log 2>/dev/null), node $(sed -n 's/^export OPD_NODE=//p' $O/paths.env)"
tail -n 3 $O/logs/heartbeat.log 2>/dev/null | sed -E 's/hf_[A-Za-z0-9]+/hf_***/g' | cut -c1-200; tail -n 2 $O/logs/heartbeat.err 2>/dev/null | cut -c1-200
echo "== later, read only: tail -n 3 $O/logs/heartbeat.log; tail -n 3 $O/logs/run_tasks.log; cat $O/control/last_pull.json ==" )
