for p in $(nvidia-smi --query-compute-apps=pid --format=csv,noheader); do kill -KILL $p 2>/dev/null; done

for p in $(pgrep -f "^ray::"); do tr '\0' '\n' < /proc/$p/environ 2>/dev/null | grep -q "ray_gpu" && kill -KILL $p 2>/dev/null; done
for i in $(seq 1 9); do G=$(nvidia-smi --query-compute-apps=pid --format=csv,noheader | grep -c .); [ "$G" = 0 ] && break; sleep 10; done
echo "GPU processes now: $G; memory used per card: $(nvidia-smi --query-gpu=memory.used --format=csv,noheader | tr '\n' ' ')"
rm -rf /tmp/ray_gpu* 2>/dev/null
echo "== 0. the 10-30 run, read only (copies kept as *.1030) =="
echo "-- /tmp/g2.txt:"; cat /tmp/g2.txt 2>/dev/null || echo "(missing)"

echo "== 1. nothing of ours running, cards idle =="
P=$(pgrep -fa "^bash /root/h800_(setup|loops|fix)\.sh|scripts/(heartbeat|run_tasks|verify|run_arm|overnight|cold_start)\.sh|scripts/task\.sh" | cut -c1-120)

echo "== 2. proxy (values not printed) =="
[ -f /root/h800_proxy.env ] || { echo "STOP: /root/h800_proxy.env missing"; exit 1; }
. /root/h800_proxy.env
a=$([ -n "${no_proxy:-}" ] && echo set || echo MISSING); b=$([ -n "${NO_PROXY:-}" ] && echo set || echo MISSING); echo "no_proxy $a, NO_PROXY $b"
if [ $a = MISSING ] || [ $b = MISSING ]; then L="localhost,127.0.0.1,::1,$(hostname),$(hostname -I 2>/dev/null | tr -s ' ' ',' | sed 's/,$//')"
  printf 'export no_proxy=%s\nexport NO_PROXY=%s\n' "$L" "$L" >> /root/h800_proxy.env; echo "  both added to /root/h800_proxy.env (localhost, 127.0.0.1, ::1, this host's name and addresses)"; fi

git -C $RB pull -q --ff-only || { echo "STOP: git pull failed"; exit 1; }

echo "runbook $(git -C $RB log -1 --format=%h)"
echo "== 4. control.env: the 80 GB profile, before verify =="
touch $O/control.env
grep '^export ' $RB/profiles/h800_80g.env | while read -r l; do grep -q "^${l%%=*}=" $O/control.env || echo "$l" >> $O/control.env; done
echo "control.env: $(sed -n 's/^export //p' $O/control.env | tr '\n' ' ')"
$O/venv-fa/bin/python -c "import flash_attn" 2>/dev/null && echo "venv-fa imports flash_attn: the smoke runs on it (VENV=fa)" \
  || echo "venv-fa not built yet (cold start step 4 builds it): the smoke runs on the main venv (PROBE_ENV VENV=), the other keys as above"
cat > /root/h800_fix.sh <<'FIXEOF'
#!/usr/bin/env bash
set -u
T=/code/opd; O=$T/run; RB=$T/runbook
. /root/h800_proxy.env
export OPD_ROOT=$O XDG_CACHE_HOME=$T/cache UV_CACHE_DIR=$T/cache/uv UV_PYTHON_INSTALL_DIR=$T/tools/python PIP_CACHE_DIR=$T/cache/pip \
       HF_HOME=$T/cache/hf TRITON_CACHE_DIR=$T/cache/triton PATH=$T/tools/uv/bin:$PATH OVERNIGHT_TIER_A_ONLY=1 FA_CUDA_ARCHS=90 \
       PYTHONDONTWRITEBYTECODE=1 VLLM_NO_USAGE_STATS=1
$O/venv-fa/bin/python -c "import flash_attn" 2>/dev/null || export PROBE_ENV="VENV="
say() { echo "[fix $(date -u +%H:%MZ)] $*"; }

  cat /tmp/g2.txt; tail -n 40 $O/logs/smoke.log | cut -c1-240; grep -nE "Error|Traceback|OutOfMemory|CUDA out of memory|ray" $O/logs/smoke.log | tail -20 | cut -c1-240
fi
FIXEOF
chmod +x /root/h800_fix.sh
tmux kill-session -t h800fix 2>/dev/null
tmux new-session -d -s h800fix 'bash /root/h800_fix.sh 2>&1 | tee -a /root/h800_fix.log; sleep infinity'
