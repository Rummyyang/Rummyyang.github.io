( set -u
cat > /root/h800_loops.sh <<'LOOPSEOF'
#!/usr/bin/env bash
# H800: after the setup prints "done: health", start the control loops with dispatch frozen (every queued task gets .paused;
# no model prefetch). Only heartbeat + git pull/control + uploader + watchdog run. Written by the executor 10-30 (code-checked, not rehearsed); v2 10-30: the setup's isolation exports, the token check.
set -u
LOG=/root/h800_setup.log; TOKEN_FILE=${TOKEN_FILE:-/root/.hf_token_opd}   # must be the setup's own TOKEN_FILE
[ -s "$TOKEN_FILE" ] || { echo "[loops] $TOKEN_FILE missing or empty: not starting loops"; exit 1; }
echo "[loops $(date -u +%H:%MZ)] waiting for the setup to print done ..."
# decide by the LAST status line of the setup log (an appended log may hold an earlier STOP or done): wait while it is not "done"
n=0; while :; do last=$(grep -E "^\[h800.*(STOP:|done: health|move: from|fresh install)" $LOG 2>/dev/null | tail -1); case "$last" in *"done: health"*) break;; *STOP:*) [ $((n % 10)) = 0 ] && echo "[loops] last setup status is STOP; waiting for a new run to finish";; esac; n=$((n+1)); [ $n -gt 480 ] && { echo "[loops] waited 8 h without a done line; giving up"; exit 1; }; sleep 60; done
T=$(sed -n "s/.*this node's files under \([^ ]*\) (.*/\1/p" $LOG | tail -1)
[ -n "$T" ] && [ -d "$T/runbook" ] || { echo "[loops] cannot find the node dir in $LOG"; exit 1; }
echo "[loops $(date -u +%H:%MZ)] setup done; node dir $T; freezing dispatch and starting loops"
cd $T/runbook || exit 1
for t in $(awk '$1 !~ /^#/ && $1 != "version:" && NF > 2 {for (i = 3; i <= NF; i++) {split($i, a, ":"); print a[1]}}' tasks/queue.txt | sort -u); do mkdir -p $T/run/runs/$t; [ -f $T/run/runs/$t/.done ] || echo "seq 0 $(date -u +%FT%TZ) (first start: dispatch frozen)" > $T/run/runs/$t/.paused; done
grep -qx 'export PREFETCH_PAR=0' $T/run/control.env || echo 'export PREFETCH_PAR=0' >> $T/run/control.env
export OPD_ROOT=$T/run HF_REPO=yang31210999/opd-lab-b300-pub HF_TOKEN=$(head -1 $TOKEN_FILE | tr -d '[:space:]')
# the setup's isolation (its step 3): caches, uv, Python and the HF home inside $T, nothing in $HOME; no .pyc next to a linked
# venv on the share (the move branch); vLLM tier A only, flash-attn for sm_90 - the loops and every task they start inherit these
export XDG_CACHE_HOME=$T/cache UV_CACHE_DIR=$T/cache/uv UV_PYTHON_INSTALL_DIR=$T/tools/python PIP_CACHE_DIR=$T/cache/pip \
       HF_HOME=$T/cache/hf TRITON_CACHE_DIR=$T/cache/triton PATH=$T/tools/uv/bin:$PATH OVERNIGHT_TIER_A_ONLY=1 FA_CUDA_ARCHS=90 \
       PYTHONDONTWRITEBYTECODE=1 VLLM_NO_USAGE_STATS=1
setsid nohup bash scripts/heartbeat.sh > $OPD_ROOT/logs/heartbeat.log 2>&1 < /dev/null &
setsid nohup bash scripts/run_tasks.sh > $OPD_ROOT/logs/run_tasks.log 2>&1 < /dev/null &
sleep 90
echo "[loops $(date -u +%H:%MZ)] paused $(ls $T/run/runs/*/.paused 2>/dev/null | wc -l), loops $(pgrep -fc 'scripts/(heartbeat|run_tasks).sh'), started $(grep -c '^START' $OPD_ROOT/logs/run_tasks.log 2>/dev/null), node $(sed -n 's/^export OPD_NODE=//p' $OPD_ROOT/paths.env 2>/dev/null)"
LOOPSEOF
chmod +x /root/h800_loops.sh
tmux kill-session -t h800loops 2>/dev/null
tmux new-session -d -s h800loops 'TOKEN_FILE=/root/.hf_token_opd bash /root/h800_loops.sh 2>&1 | tee -a /root/h800_loops.log; sleep infinity'
sleep 3; echo "== armed =="; tail -n 3 /root/h800_loops.log
echo "== later: tail -n 5 /root/h800_loops.log == (expected after the setup finishes: paused ~700, loops >= 2, started 0)" )
